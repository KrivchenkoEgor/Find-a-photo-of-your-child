# requirements.txt — единственный источник правды по зависимостям.
#
# Проверено 2026-09-28 на macOS arm64, Python 3.11.15: пакеты не просто ставятся,
# а реально находят лицо на тестовом фото (см. check_env.py и check_models.py).
# Старые пины 2024 года (PySide6 6.6.3, insightface 0.7.3, numpy 1.26.4) под этот
# Python не подходили: 4 из 8 пакетов не устанавливались вовсе.
#
# Установка (два шага, порядок важен):
#   uv venv --python /opt/homebrew/opt/python@3.11/bin/python3.11 .venv
#   uv pip install --python .venv/bin/python PySide6==6.11.2 opencv-python-headless==5.0.0.93 \
#       numpy==2.4.6 onnxruntime==1.30.0 onnx==1.23.0 scipy==1.17.1 scikit-image==0.26.0 \
#       pillow-heif==1.8.0 Pillow==12.3.0 tqdm==4.70.1 requests==2.34.2
#   uv pip install --python .venv/bin/python --no-deps insightface==2.0
#
# Почему insightface ставится отдельно с --no-deps: он заявляет зависимость
# opencv-python, а opencv-python и opencv-python-headless укладывают свой код
# в один и тот же каталог cv2 и затирают друг друга. Все остальные зависимости
# insightface перечислены выше вручную.

# --- Распознавание лиц (режим «Конкретный человек») ---

pytest==9.1.1
